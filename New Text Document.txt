<script>
const DB_RESOURCES = [
  {id:1,name:"CS Lab AB1-301",cat:"lab",emoji:"[CPU]",loc:"AB1-F3",fill:30},
  {id:2,name:"CS Lab AB2-202",cat:"lab",emoji:"[CPU]",loc:"AB2-F2",fill:65},
  {id:5,name:"Main Library",cat:"lib",emoji:"[DB]",loc:"AB1-G",fill:40},
  {id:7,name:"Audi 1",cat:"audi",emoji:"[AMP]",loc:"AB1-G",fill:10},
];

const slots=["09:00","10:00","11:00","12:00","13:00","14:00","15:00","16:00","17:00"];
let myBookings = [], selRes=null, selSlot=null;

function showView(v,btn){
  document.querySelectorAll('.view').forEach(el=>el.style.display='none');
  document.querySelectorAll('.nav-btn').forEach(el=>el.classList.remove('active'));
  document.getElementById('view-'+v).style.display='grid';
  btn.classList.add('active');
  if(v==='mybookings') renderMyBookings();
}

function renderGrid(containerId, list){
  document.getElementById(containerId).innerHTML = list.map(r=>`
    <div class="r-card ${selRes?.id===r.id?'selected':''}" onclick="selectRes(${r.id})">
      <div style="color:var(--neon-blue); font-size:10px;">${r.emoji}</div>
      <div style="font-family:Orbitron; font-size:12px; margin:5px 0;">${r.name}</div>
      <div style="font-size:9px; color:#666;">${r.loc} | LOAD: ${r.fill}%</div>
      <div style="height:2px; background:#222; margin-top:8px;"><div style="width:${r.fill}%; height:100%; background:var(--neon-purple);"></div></div>
    </div>
  `).join('');
}

function selectRes(id){
  selRes = DB_RESOURCES.find(r=>r.id===id);
  document.getElementById('bp-res-name').textContent = selRes.name;
  renderSlots();
  document.querySelectorAll('.r-card').forEach(c=>c.classList.remove('selected'));
}

function renderSlots(){
  document.getElementById('slot-grid').innerHTML = slots.map(s=>`
    <div class="slot ${selSlot===s?'sel':''}" onclick="selSlot='${s}';renderSlots()">${s}</div>
  `).join('');
}

function confirmBooking(){
  if(!selRes || !selSlot) return;
  myBookings.push({ id: Math.random().toString(16).slice(2,8).toUpperCase(), res: selRes.name, slot: selSlot });
  document.getElementById('alert-success').style.display='block';
  document.getElementById('s-mine').textContent = myBookings.length;
  setTimeout(()=>document.getElementById('alert-success').style.display='none', 2000);
}

function renderMyBookings(){
  document.getElementById('mb-list').innerHTML = myBookings.map(b=>`
    <div class="r-card" style="display:flex; justify-content:space-between; align-items:center;">
      <div>
        <div style="color:var(--neon-blue); font-family:Orbitron;">${b.res}</div>
        <div style="font-size:10px;">TIME: ${b.slot} | AUTH: ${b.id}</div>
      </div>
      <button class="nav-btn" style="border-color:var(--neon-pink); color:var(--neon-pink)" onclick="cancel('${b.id}')">TERMINATE</button>
    </div>
  `).join('') || '<div style="color:#444">NO_ACTIVE_RESERVATIONS</div>';
}

function cancel(id){
  myBookings = myBookings.filter(b=>b.id!==id);
  renderMyBookings();
  document.getElementById('s-mine').textContent = myBookings.length;
}

function switchTab(cat, btn){
  renderGrid('book-grid', DB_RESOURCES.filter(r=>r.cat===cat));
  document.querySelectorAll('#view-book .nav-btn').forEach(b=>b.classList.remove('active'));
  btn.classList.add('active');
}

renderGrid('home-grid', DB_RESOURCES);
renderGrid('book-grid', DB_RESOURCES.filter(r=>r.cat==='lab'));
renderSlots();
</script>
